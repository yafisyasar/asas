"""Reading XML in python

For reading the data from XML file you can import xml.etree. ElementTree library.
"""

import xml.etree.ElementTree as ET
tree = ET.parse('/content/testxml.xml')
root = tree.getroot() # this points to the root node
print(root.tag)# printing the root tag name
#iterate over subelements (commonly called "children") in the root by using a simple "for" loop.
for child in root:
    print(child.tag, child.attrib)

# reading the xml file
print(ET.tostring(root, encoding='utf8').decode('utf8'))

for company in root.iter('company'):
    print(company.text)
